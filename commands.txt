# Instala as bibliotecas novas do requirements.txt
pip install -r requirements.txt

# Instalar o Django 5
pip install django==5.2.12

# Rodar o projeto
python manage.py runserver

# Conferir o models e preparar o SQL para o banco
python manage.py makemigrations

# Usar o SQL criado para efetivamente atualizar o banco
python manage.py migrate

# Criar um superusuário no banco: admin/admin
python manage.py createsuperuser

# Instalar a biblioteca Django Crispy Forms
pip install django-crispy-forms

# Roda os testes reaproveitando o banco de testes do Neon (sem a pergunta do "yes")
python manage.py test --keepdb

# ===== Uma vez por máquina: preparar o ngrok (bot do Telegram) =====
winget install ngrok.ngrok
ngrok config add-authtoken SEU_TOKEN_DO_NGROK
# Se o terminal disser que não conhece o ngrok, rode esta linha no terminal antes de usá-lo
$env:Path += ";$env:LOCALAPPDATA\Microsoft\WinGet\Packages\Ngrok.Ngrok_Microsoft.Winget.Source_8wekyb3d8bbwe"

# ===== Para testar o bot pelo celular (dois terminais) =====
# Terminal 1
python manage.py runserver
# Terminal 2: túnel público para o Telegram chegar ao seu computador
ngrok http 8000

# Só se o endereço do ngrok mudar: atualize ALLOWED_HOSTS, SITE_URL e
# TELEGRAM_BOT_WEBHOOK_URL no .env, reinicie o runserver e registre o webhook de novo
python manage.py telegram_webhook

# Antes do deploy gerar os arquivos estáticos com o comando abaixo
python manage.py collectstatic

# Comando para fazer o deploy no Google Cloud
gcloud app deploy
